cmake_minimum_required(VERSION 3.20)
project(ForestFrontier VERSION 0.9 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(frontier_core src/world.cpp src/perf.cpp src/gameplay.cpp src/content.cpp src/save_slots.cpp src/adventure.cpp src/summary.cpp src/controls.cpp src/progress.cpp src/geometry.cpp src/atlas_notes.cpp src/vessels.cpp src/settlement_system.cpp src/nation.cpp src/nation_save.cpp src/resident_orders.cpp src/freight.cpp src/work_scheduler.cpp)
target_compile_definitions(frontier_core PUBLIC FRONTIER_ROOT="${CMAKE_SOURCE_DIR}")
target_include_directories(frontier_core PUBLIC src third_party)
target_link_libraries(frontier_core PUBLIC Threads::Threads)
target_compile_options(frontier_core PRIVATE -Wall -Wextra -Wpedantic)

option(FRONTIER_BUILD_GAME "Build the graphical game" ON)
if(FRONTIER_BUILD_GAME)
  find_package(raylib 5 QUIET)
  if(NOT raylib_FOUND)
    if(APPLE AND EXISTS "${CMAKE_SOURCE_DIR}/third_party/raylib-5.5/CMakeLists.txt")
      enable_language(C)
      set(BUILD_EXAMPLES OFF CACHE BOOL "Build raylib examples" FORCE)
      set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build shared libraries" FORCE)
      add_subdirectory(third_party/raylib-5.5 EXCLUDE_FROM_ALL)
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
      set(RAYLIB_DIR "${CMAKE_SOURCE_DIR}/third_party/raylib-5.5_linux_amd64")
      if(NOT EXISTS "${RAYLIB_DIR}/lib/libraylib.a")
        message(FATAL_ERROR "raylib missing. Run ./scripts/bootstrap.sh first.")
      endif()
      add_library(raylib STATIC IMPORTED)
      set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${RAYLIB_DIR}/lib/libraylib.a"
        INTERFACE_INCLUDE_DIRECTORIES "${RAYLIB_DIR}/include"
        INTERFACE_LINK_LIBRARIES "Threads::Threads;${CMAKE_DL_LIBS};m")
    else()
      message(FATAL_ERROR "Native raylib >=5 missing. Run ./scripts/bootstrap.sh on macOS, or install raylib for your target platform.")
    endif()
  endif()
  add_library(frontier_dialogue src/dialogue.cpp)
  target_include_directories(frontier_dialogue PUBLIC src third_party)
  target_link_libraries(frontier_dialogue PUBLIC Threads::Threads)
  target_compile_options(frontier_dialogue PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(forest_frontier src/main.cpp src/render.cpp src/cartography.cpp src/menu.cpp src/summary_render.cpp src/input.cpp src/settings_ui.cpp src/audio.cpp src/map_journal.cpp src/nation_ui.cpp)
  target_link_libraries(forest_frontier PRIVATE frontier_core frontier_dialogue raylib)
  target_compile_definitions(forest_frontier PRIVATE FRONTIER_ROOT="${CMAKE_SOURCE_DIR}")
  target_compile_options(forest_frontier PRIVATE -Wall -Wextra -Wpedantic)
endif()
