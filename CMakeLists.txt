cmake_minimum_required(VERSION 3.20)
project(ForestFrontier VERSION 0.6 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(frontier_core src/world.cpp src/perf.cpp src/gameplay.cpp src/content.cpp src/save_slots.cpp src/adventure.cpp src/summary.cpp)
target_compile_definitions(frontier_core PUBLIC FRONTIER_ROOT="${CMAKE_SOURCE_DIR}")
target_include_directories(frontier_core PUBLIC src third_party)
target_link_libraries(frontier_core PUBLIC Threads::Threads)
target_compile_options(frontier_core PRIVATE -Wall -Wextra -Wpedantic)

option(FRONTIER_BUILD_GAME "Build the graphical game" ON)
if(FRONTIER_BUILD_GAME)
  find_package(raylib 5 QUIET)
  if(NOT raylib_FOUND)
    set(RAYLIB_DIR "${CMAKE_SOURCE_DIR}/third_party/raylib-5.5_linux_amd64")
    if(NOT EXISTS "${RAYLIB_DIR}/lib/libraylib.a")
      message(FATAL_ERROR "raylib missing. Run ./scripts/bootstrap.sh first.")
    endif()
    add_library(raylib STATIC IMPORTED)
    set_target_properties(raylib PROPERTIES IMPORTED_LOCATION "${RAYLIB_DIR}/lib/libraylib.a"
      INTERFACE_INCLUDE_DIRECTORIES "${RAYLIB_DIR}/include"
      INTERFACE_LINK_LIBRARIES "Threads::Threads;${CMAKE_DL_LIBS};m")
  endif()
  add_library(frontier_dialogue src/dialogue.cpp)
  target_include_directories(frontier_dialogue PUBLIC src third_party)
  target_link_libraries(frontier_dialogue PUBLIC Threads::Threads)
  target_compile_options(frontier_dialogue PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(forest_frontier src/main.cpp src/render.cpp src/cartography.cpp src/menu.cpp src/summary_render.cpp)
  target_link_libraries(forest_frontier PRIVATE frontier_core frontier_dialogue raylib)
  target_compile_definitions(forest_frontier PRIVATE FRONTIER_ROOT="${CMAKE_SOURCE_DIR}")
  target_compile_options(forest_frontier PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(summary_tests tests/summary_tests.cpp)
  target_link_libraries(summary_tests PRIVATE frontier_core)
  add_test(NAME summary COMMAND summary_tests)
  add_executable(quests_tests tests/quests_tests.cpp)
  target_link_libraries(quests_tests PRIVATE frontier_core)
  add_test(NAME quests COMMAND quests_tests)
  add_executable(save_slots_tests tests/save_slots_tests.cpp)
  target_link_libraries(save_slots_tests PRIVATE frontier_core)
  add_test(NAME save_slots COMMAND save_slots_tests)
  add_executable(world_tests tests/world_tests.cpp)
  target_link_libraries(world_tests PRIVATE frontier_core)
  add_test(NAME world COMMAND world_tests)
  add_executable(gameplay_tests tests/gameplay_tests.cpp)
  target_link_libraries(gameplay_tests PRIVATE frontier_core)
  add_test(NAME gameplay COMMAND gameplay_tests)
  add_executable(campaign_tests tests/campaign_tests.cpp)
  target_link_libraries(campaign_tests PRIVATE frontier_core)
  add_test(NAME campaign COMMAND campaign_tests)
  add_executable(adventure_tests tests/adventure_tests.cpp)
  target_link_libraries(adventure_tests PRIVATE frontier_core)
  add_test(NAME adventure COMMAND adventure_tests)
  if(FRONTIER_BUILD_GAME)
    add_executable(dialogue_probe tests/dialogue_probe.cpp)
    target_link_libraries(dialogue_probe PRIVATE frontier_dialogue)
    find_package(Python3 COMPONENTS Interpreter)
    if(Python3_FOUND)
      add_test(NAME dialogue COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/dialogue_test.py $<TARGET_FILE:dialogue_probe>)
      if(UNIX)
        find_program(FRONTIER_CURL curl)
        if(FRONTIER_CURL)
          add_test(NAME launcher COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/launcher_test.py)
          set_tests_properties(launcher PROPERTIES TIMEOUT 90)
        endif()
      endif()
    endif()
  endif()
endif()
